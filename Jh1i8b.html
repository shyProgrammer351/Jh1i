문제

미로 맵에서 한 칸 이동하는 문제입니다 이를 해결하는 다음 프로그램에 대해 빈칸을 채우세요 
+  O   +
 [ 옷 ->
+  문  +
WASD와 [Enter]를 입력하세요: a
방패를 획득했습니다 


+  O   +
 [ 옷 ->
+  문  +
WASD와 [Enter]를 입력하세요: W
물약을 획득했습니다 


+  O   +
 [ 옷 ->
+  문  +
WASD와 [Enter]를 입력하세요: s
출구로 탈출합니다

코드 빈칸 채우기

 java.util.Scanner; 
 
	typedef enum { UP = 'w', DOWN = 's', LEFT = 'a', RIGHT = 'd' } Direction;	  
 
public class Main 
{ 
	public static void main( String[] args ) { 
		Scanner scan = new Scanner( System.in ); 
		char move = '\0'; 
 
		System.out.println( "+  O   +" ); 
		System.out.println( " [ [34m옷 [0m->" ); 
		System.out.println( "+  문  +" ); 
		 
		System.out.print( "WASD와 [Enter]를 입력하세요: " ); 
		move= scan.nextLine().charAt(0); 
	 
		( move ) { 
			 UP: 
				System.out.print( "물약을 획득했습니다" ); 
				; 
			 LEFT: 
				System.out.print( "방패를 획득했습니다" ); 
				; 
			 DOWN: 
				System.out.print( "출구로 탈출합니다" ); 
				; 
			 RIGHT: 
				System.out.print( "무기를 획득했습니다" ); 
				; 
			: 
				System.out.print( "WASD를 입력하세요" ); 
		} 
	} 
}
