알고리즘

프로그램 시작 
   사용자와 시스템의 대화 반복 
      사용자 입력 
      시스템 답변 출력 
         사용자가 '바이'를 입력하면 대화 종료 
         준비한 대화쌍에서 
            적절한 답변을 찾으면 시스템 답변을 출력 
            적절한 답변을 못 찾으면 '뭐 딴거 재밌는거 없을까?' 출력 
프로그램 종료

프로그램 코드 

	// 파일명 : ./Chapter15/Chat.java

	#include <string.h>

	 

		// 문자열 최대 길이 정의

		#define MAX_STR_LEN	1024

	 

		// 사용자와 시스템의 대화쌍

		typedef struct {

			String	user;

			String	system;

		} Pair;

	 

	public class Main

	{

		// 프로그램 시작 	

1		public static void main( String[] args ) {

			int  index = 0;

			String input;

			Pair dialogue[4] =

				{ { "이름이 뭐", "난 재석이야 국민MC 유재석과 이름이 같아 ㅎㅎㅎ 넌 몇 살이니?" },

				  { "몇 살", "난 20살이야 남들이 다 부러워하는 나이지 넌 어디 살아?" },

				  { "어디 살", "난 홍지동 살아 북한산 아래 있어서 공기가 무지 맑아ㅎㅎㅎ 취미가 뭐니?" },

				  { "만나서 반가", "나도 반가워 친하게 지내자" } };

		 

			// 사용자와 시스템의 대화 반복 

2			System.out.print( "[시스템] 안녕? 넌 이름이 모야?\n 사용자: " );

3			while( 1 ) {

				// 사용자 입력 	

4				fgets( input, MAX_STR_LEN, stdin );

		 

				// 시스템 답변 출력			

				// 사용자가 '바이'를 입력하면 대화 종료			

5				if ( strstr( input, "바이" ) != NULL ) {

					System.out.println( "[시스템] 그래 안녕! 담에 또 봐~" );

6					break;

				}

				 

				// 준비한 대화쌍에서   

7				for ( index = 0; index < 4; index++ ) {

					// 적절한 답변을 찾으면 시스템 답변을 출력  

8					if ( strstr( input, dialogue[index].user ) != NULL ) {

						System.out.print( "[시스템] " + dialogue[index].system + "\n 사용자: " );

9						break;

					}

				}

				 

				// 적절한 답변을 못 찾으면 '뭐 딴거 재밌는거 없을까?' 출력 

10				if ( index == 4 ) {

11					System.out.print( "[시스템] 뭐 딴거 재밌는거 없을까?\n 사용자: " );

				}

			}

	 

		// 프로그램 종료 	

12		}

	}
